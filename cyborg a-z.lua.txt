															pcall(function()
																for _, child in pairs(LocalPlayer.Backpack:GetChildren()) do
																	if
																		child:IsA("Tool")
																		and (child.ToolTip == "Blox Fruit" or child.Name:lower():find("fruit"))
																	then
																		COMMF_:InvokeServer("StoreFruit", child.Name)
																		task.wait(0.5)
																	end
																end

																if Character then
																	for _, child in pairs(Character:GetChildren()) do
																		if
																			child:IsA("Tool")
																			and (child.ToolTip == "Blox Fruit" or child.Name:lower():find("fruit"))
																		then
																			COMMF_:InvokeServer("StoreFruit", child.Name)
																			task.wait(0.5)
																		end
																	end
																end
															end)

															return
														end
													end
												end
											end
										end

										return
									end

									SetStatus("Kaitun Get Cyborg A-Z")
									task.wait(1.5)
									SetStatus("Travel to sea 2")
									task.wait(1)

									if currentMapIdx ~= 1 then
										if currentMapIdx == 3 then
											COMMF_:InvokeServer("TravelDressrosa")
										end

										return
									end

									COMMF_:InvokeServer("TravelDressrosa")
								end, function(err)
									warn(err)
									StarterGui:SetCore("SendNotification", {
										Title = "ERROR",
										Text = err,
									})
								end)
							end
						end)
						task.spawn(function()
							while task.wait(4) do
								xpcall(function()
									if Character.Humanoid and not (Character.Humanoid.Health <= 0) then
										if not Character:FindFirstChild("HasBuso") then
											COMMF_:InvokeServer("Buso")
										end

										for _, v in
											next,
											{
												"Buso",
												"Geppo",
												"Soru",
											}
										do
											if
												not CollectionService:HasTag(Character, v)
												and LocalPlayer.Data.Beli.Value
													>= (function(skillName)
														if skillName ~= "Geppo" then
															if skillName ~= "Buso" then
																if skillName ~= "Soru" then
																	return 0
																end

																return 100000
															end

															return 25000
														end

														return 10000
													end)(v)
											then
												COMMF_:InvokeServer("BuyHaki", v)
											end
										end

										return
									end
								end, function(err)
									warn("LL: " .. err)
								end)
							end
						end)

						local ErrorMessageChanged = GuiService.ErrorMessageChanged
						local packedErrorHookArgs = PackArgs(newcclosure(function()
							if GuiService:GetErrorType() ~= Enum.ConnectionError.DisconnectErrors then
								return
							end

							while true do
								TeleportService:TeleportToPlaceInstance(PlaceId, JobId, LocalPlayer)
								task.wait(5)
							end
						end))

						ErrorMessageChanged:Connect(unpack(packedErrorHookArgs, 1, packedErrorHookArgs.n))

						return
					end

					workspaceRef = cloneref(Workspace)

					if workspaceRef then
						workspaceFound = true
					end
				until not workspaceFound

				if getrenv then
					workspaceRef = getrenv().workspace

					if workspaceRef then
						workspaceFound = true
					end

					if not workspaceFound then
						workspaceRef = getrenv().Workspace

						if workspaceRef then
							workspaceFound = true
						end
					end
				end
			until not workspaceFound
		until not workspaceFound
		local packedWorkspaceArgs = PackArgs(game:GetService("Workspace"))

		workspaceRef = cloneref(unpack(packedWorkspaceArgs, 1, packedWorkspaceArgs.n))
		workspaceFound = true
	until not workspaceFound
end
